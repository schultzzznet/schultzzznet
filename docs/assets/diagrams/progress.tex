% ---------------------------------------------------------------- geometry
\newcommand{\W}{30}          % picture width
\newcommand{\TileW}{5.6}     % headline tile
\newcommand{\PhW}{5.0}       % phase chevron pitch
\newcommand{\StepGap}{0.2}

\newcommand{\DrillU}{0.42}    % cm per drill
\newcommand{\RuleU}{0.52}    % cm per rule
\newcommand{\RuleX}{14.8}    % where the rule bars start

% A headline tile: index, value, of, label, sub line
\newcommand{\ProgressTile}[5]{%
  \pgfmathsetmacro{\x}{#1*(\TileW+0.5)}
  \draw[draw=bandline, fill=white, rounded corners=4pt, line width=0.7pt] (\x,25.3) rectangle ++(\TileW,3.75);
  \node[anchor=north west, inner sep=0pt] at (\x+0.3,28.75) {{\fontsize{30}{32}\selectfont\bfseries #2}\,{\Large\color{ink!60} of #3}};
  \node[anchor=north west, inner sep=0pt, font=\small\bfseries, text width=5.0cm, align=left] (lbl) at (\x+0.3,27.5) {#4};
  \node[anchor=north west, inner sep=0pt, font=\footnotesize, text=ink!70, text width=5.0cm, align=left] at ([yshift=-1.2mm]lbl.south west) {#5};
  \fill[band] (\x+0.3,25.55) rectangle ++(\TileW-0.6,0.2);
  \pgfmathsetmacro{\f}{(\TileW-0.6)*#2/#3}
  \ifdim\f pt>0pt \fill[multi] (\x+0.3,25.55) rectangle ++(\f,0.2);\fi}

% The bracket over the phases of one group: group, first phase, last phase
\newcommand{\ProgressGroup}[3]{%
  \pgfmathsetmacro{\a}{#2*\PhW+0.15}
  \pgfmathsetmacro{\b}{(#3+1)*\PhW-0.25}
  \draw[ink!45, line width=0.8pt] (\a,23.75) -- ++(0,0.22) -- (\b,23.97) -- ++(0,-0.22);
  \node[fill=white, inner xsep=5pt, font=\bfseries, text=ink!75] at ($(\a,23.97)!0.5!(\b,23.97)$) {\MakeUppercase{#1}};}

% A phase chevron and its note: number, state, title, note
\newcommand{\ProgressPhase}[4]{%
  \pgfmathsetmacro{\x}{#1*\PhW}
  \pgfmathsetmacro{\n}{#1>0 ? 0.45 : 0}
  \draw[phase #2, line join=round]
    (\x,23.2) -- ++(\PhW-0.3,0) -- ++(0.45,-1.3) -- ++(-0.45,-1.3) -- ++(-\PhW+0.3,0) -- ++(\n,1.3) -- cycle;
  \node[anchor=north west, inner sep=0pt, font=\scriptsize\bfseries, pink #2] at (\x+0.6,22.95) {PHASE #1};
  \node[anchor=north west, inner sep=0pt, font=\small\bfseries, pink #2, text width=3.9cm, align=left] at (\x+0.6,22.5) {#3};
  \node[anchor=south west, inner sep=0pt, font=\scriptsize\bfseries, pstc #2] at (\x+0.6,20.85) {\StateName{#2}};
  \ifnum\StrCmp{#2}{current}=0
    \node[pill, fill=pull, text=white, font=\sffamily\bfseries\small, inner xsep=6pt] at (\x+2.3,23.32) {WE ARE HERE};
  \fi
  \node[anchor=north west, inner sep=0pt, font=\footnotesize, text=ink!80, text width=4.45cm, align=left] at (\x+0.15,20.3) {#4};}

% A step tile: number, state, title
\newcommand{\ProgressStep}[3]{%
  \pgfmathsetmacro{\x}{#1*(\StepW+\StepGap)}
  \draw[step #2, rounded corners=4pt] (\x,12.75) rectangle ++(\StepW,3.5);
  \node[anchor=north west, inner sep=0pt, font=\bfseries\fontsize{22}{24}\selectfont, ink #2] at (\x+0.2,16.0) {#1};
  \node[anchor=north west, inner sep=0pt, font=\footnotesize\bfseries, ink #2, text width=\StepW cm-0.35cm, align=left] at (\x+0.2,15.0) {#3};
  \node[anchor=south west, inner sep=0pt, font=\fontsize{6.5}{7.5}\selectfont\bfseries, stc #2, text width=\StepW cm-0.3cm, align=left] at (\x+0.2,12.95) {\StateName{#2}};}

% A row of drills: index, kind, live|simulated, dates, run, passed
\newcommand{\ProgressDrill}[6]{%
  \pgfmathsetmacro{\y}{9.75-#1*1.45}
  \ifnum\StrCmp{#3}{simulated}=0
    \node[anchor=west, inner sep=0pt] at (0,\y) {{\small\bfseries #2}\enspace{\footnotesize\color{simc}\bfseries simulated, #4}};
  \else
    \node[anchor=west, inner sep=0pt] at (0,\y) {{\small\bfseries #2}\enspace{\footnotesize\color{ink!65}home site, #4}};
  \fi
  \foreach \k in {1,...,#5} {
    \pgfmathtruncatemacro{\ok}{\k<=#6 ? 1 : 0}
    \ifnum\StrCmp{#3}{simulated}=0
      \ifnum\ok=1 \def\st{simpass}\else\def\st{simfail}\fi
    \else
      \ifnum\ok=1 \def\st{pass}\else\def\st{fail}\fi
    \fi
    \draw[drill \st, rounded corners=1pt] (\k*\DrillU-\DrillU,\y-0.75) rectangle ++(\DrillU-0.08,0.42);}
  \pgfmathtruncatemacro{\failed}{#5-#6}
  \node[anchor=west, inner sep=0pt, font=\footnotesize] at (#5*\DrillU+0.15,\y-0.54)
    {\textcolor{multi}{\bfseries #6 passed}\enspace\textcolor{ink!50}{/}\enspace\textcolor{failc}{\bfseries \failed\ not passed}};}

% A rule bar: index, name, rules, built, partly, not built
\newcommand{\ProgressRule}[6]{%
  \pgfmathsetmacro{\y}{9.75-#1*0.56}
  \node[anchor=east, inner sep=0pt, font=\footnotesize] at (\RuleX-0.2,\y) {#2};
  \foreach \k in {1,...,#3} {
    \pgfmathtruncatemacro{\c}{\k<=#4 ? 1 : (\k<=#4+#5 ? 2 : 3)}
    \draw[rule\c, rounded corners=1pt] (\RuleX+\k*\RuleU-\RuleU,\y-0.18) rectangle ++(\RuleU-0.08,0.36);}
  \node[anchor=west, inner sep=0pt, font=\scriptsize] at (\RuleX+#3*\RuleU+0.15,\y)
    {\textcolor{multi}{\bfseries #4}\,\textcolor{ink!50}{/}\,\textcolor{partc!85!black}{\bfseries #5}\,\textcolor{ink!50}{/}\,\textcolor{ink!60}{#6}\enspace\textcolor{ink!50}{of #3}};}
